cmake_minimum_required(VERSION 3.24)
project(tech-news VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(TECH_NEWS_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(CURL 7.85 REQUIRED)
find_package(LibXml2 2.9.10 REQUIRED)
find_package(SQLite3 3.24 REQUIRED)
find_package(Threads REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(TECH_NEWS_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(TECH_NEWS_SQLITE_TARGET SQLite::SQLite3)
endif()

include(FetchContent)
set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ftxui
  URL https://codeload.github.com/ArthurSonzogni/FTXUI/tar.gz/refs/tags/v7.0.3
  URL_HASH SHA256=e7c62ffe19009759821b4f0f8df7f2a6fb83784c3a9f1477d81f56d3ee723c88
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(ftxui)

add_library(tech_news_core
  src/text.cpp src/parser.cpp src/http.cpp src/store.cpp src/worker.cpp src/platform.cpp)
target_include_directories(tech_news_core PUBLIC include)
target_link_libraries(tech_news_core PUBLIC CURL::libcurl LibXml2::LibXml2 ${TECH_NEWS_SQLITE_TARGET} Threads::Threads)
target_compile_options(tech_news_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

add_executable(tech-news src/main.cpp src/ui.cpp)
target_link_libraries(tech-news PRIVATE tech_news_core ftxui::component ftxui::dom ftxui::screen)
target_compile_definitions(tech-news PRIVATE TECH_NEWS_VERSION="${PROJECT_VERSION}")
target_compile_options(tech-news PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

function(tech_news_sanitize target)
  if(TECH_NEWS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
tech_news_sanitize(tech_news_core)
tech_news_sanitize(tech-news)

if(BUILD_TESTING)
  add_executable(tech-news-tests tests/core_tests.cpp)
  target_link_libraries(tech-news-tests PRIVATE tech_news_core)
  target_compile_definitions(tech-news-tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_compile_options(tech-news-tests PRIVATE -Wall -Wextra -Wpedantic)
  tech_news_sanitize(tech-news-tests)
  add_test(NAME core COMMAND tech-news-tests)
  set_tests_properties(core PROPERTIES TIMEOUT 30)
  find_package(Python3 3.9 COMPONENTS Interpreter REQUIRED)
  add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py $<TARGET_FILE:tech-news>)
  set_tests_properties(terminal PROPERTIES TIMEOUT 45)
  add_test(NAME secret-scan COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/scan_secrets.py ${CMAKE_CURRENT_SOURCE_DIR})
endif()

include(GNUInstallDirs)
install(TARGETS tech-news RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
